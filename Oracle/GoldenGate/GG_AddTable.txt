              LOGFILE=TableName_exp.log

        - TableName_imp.par

        MAP SchemaName.TableName, TARGET SchemaName.TableName, FILTER ( @GETENV ('TRANSACTION', 'CSN') > ExportFlashbackScn);

11. import table

13. start replicat

15. update replicat parameter file: remove (comment out) the new filtered mapping command
